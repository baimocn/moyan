## 1. 新建 `frontend/src/utils/pv.js`

```js
// 墨衍小程序 · 浏览量埋点（STATS-02）：失败静默，绝不影响主流程
let _did = ''
function deviceId() {
  if (_did) return _did
  try {
    _did = uni.getStorageSync('moyan:mp:device') || ''
    if (!_did) {
      _did = 'mp' + Date.now().toString(36) + Math.random().toString(36).slice(2, 10)
      uni.setStorageSync('moyan:mp:device', _did)
    }
  } catch (e) { _did = 'anon' }
  return _did
}

export function reportPv(page) {
  try {
    uni.request({
      url: 'https://moyan.baimo7715.top/api/metrics/pv',
      method: 'POST',
      data: { source: 'mp', page: page || 'home', device_id: deviceId() },
      success: () => {}, fail: () => {},
    })
  } catch (e) { /* 静默 */ }
}
```

> 注意：生产域名已在小程序后台白名单（request 合法域名含 https://moyan.baimo7715.top），无需额外配置。
